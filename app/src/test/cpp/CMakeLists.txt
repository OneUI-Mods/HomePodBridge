cmake_minimum_required(VERSION 3.22)
project(homepod_hal_tests LANGUAGES C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_executable(hal_layout_tests hal_layout_tests.c)
if(MSVC)
    target_compile_options(hal_layout_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(hal_layout_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME owned_hal_layout_guards COMMAND hal_layout_tests)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR ANDROID)
    if(NOT ANDROID)
        find_package(JNI REQUIRED)
    endif()
    find_package(Threads REQUIRED)
    add_executable(hal_safety_tests hal_safety_tests.c)
    target_compile_options(hal_safety_tests PRIVATE -UNDEBUG)
    target_include_directories(hal_safety_tests PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(hal_safety_tests PRIVATE Threads::Threads)
    add_test(NAME ptrace_cleanup_and_signals COMMAND hal_safety_tests)
    set_tests_properties(ptrace_cleanup_and_signals PROPERTIES TIMEOUT 15)
endif()
