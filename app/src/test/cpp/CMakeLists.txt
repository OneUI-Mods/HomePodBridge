cmake_minimum_required(VERSION 3.22)
project(homepod_hal_tests LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
add_executable(hal_layout_tests hal_layout_tests.c)
if(MSVC)
    target_compile_options(hal_layout_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(hal_layout_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME owned_hal_layout_guards COMMAND hal_layout_tests)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR ANDROID)
    if(NOT ANDROID)
        find_package(JNI REQUIRED)
    endif()
    find_package(Threads REQUIRED)
    add_executable(fmq_mapping_tests fmq_mapping_tests.cpp)
    add_executable(fmq_control_tests fmq_control_tests.cpp)
    target_compile_options(fmq_control_tests PRIVATE -UNDEBUG)
    target_link_libraries(fmq_control_tests PRIVATE Threads::Threads)
    add_test(NAME fmq_standby_control COMMAND fmq_control_tests)
    set_tests_properties(fmq_standby_control PROPERTIES TIMEOUT 10)
    target_compile_options(fmq_mapping_tests PRIVATE -UNDEBUG)
    target_link_libraries(fmq_mapping_tests PRIVATE Threads::Threads)
    if(ANDROID)
        target_link_libraries(fmq_mapping_tests PRIVATE android)
        target_link_libraries(fmq_control_tests PRIVATE android)
    endif()
    add_test(NAME fmq_mapping_and_ordering COMMAND fmq_mapping_tests)
    set_tests_properties(fmq_mapping_and_ordering PROPERTIES TIMEOUT 10)
    add_executable(hal_safety_tests hal_safety_tests.c)
    target_compile_options(hal_safety_tests PRIVATE -UNDEBUG)
    target_include_directories(hal_safety_tests PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(hal_safety_tests PRIVATE Threads::Threads)
    add_test(NAME ptrace_cleanup_and_signals COMMAND hal_safety_tests)
    set_tests_properties(ptrace_cleanup_and_signals PROPERTIES TIMEOUT 15)
endif()
